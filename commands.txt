Prefix: '$'

################
### COMMANDS ###
################

$kek (alias: $bonus)
  Claim a reward and register the first time you use it.

$send <user> <amount|all> (aliases: $sendkeks, $give, $givekeks)
  Send keks to another player.

$coinflip <heads|tails> <amount|all>

$stocks (aliases: $price, $prices)
  List all stocks and their prices.

$portfolio

$buy <stock> <amount|all>
  Max 100 shares per stock

$sell <stock> <amount|all>

$holders <stock>
  List the top 10 holders of a stock.

$steal <user>
  Requires 10 keks. 35% success. Risk half your balance to steal double your balance.

$duel <user> <amount>

$accept

$eat <amount|full> or $heal <user> <amount|full> (alias: $heal)
  Heal yourself or another player. 5 keks per HP.

$hp <user (optional)>
  Show your or another player's HP bar.

$res <user>
  Resurrect another player. Costs 500 keks.

$xp <user (optional)> (alias: $level)
  Show your or another player's XP bar.

$convert <xp amount|all>
  Convert XP into keks (10 xp = 1 kek).

$stats <user (optional)> (alias: $keks)

$leaderboard (aliases: $top, $kekkers)
  10 richest players.

$bottom
  10 poorest players.

$graveyard
  List dead players and their respawn times.

$commands
  Link to this commands list.
